# Security constraints shared across environments.
# CVE-2024-23342 was reported against ecdsa, which was pulled in via python-jose.
# pip-audit lists no fix_versions for this CVE, so we hard-block ecdsa from being
# installed until a fixed version is available.
ecdsa==0

# GHSA-2h4p-vjrc-8xpq (mako, fixed in 1.3.12) and GHSA-5639-2j2p-m4mx /
# CVE-2026-102991 (mako <= 1.4.1, Windows drive-letter path traversal in
# TemplateLookup, fixed in 1.4.2). Mako is a transitive dep of alembic; pin
# here so all installs pick up the patched version.
mako==1.4.3

# P0-DEPENDENCY-REMEDIATION-001 — transitive production package floors.
# These packages are not direct deps but are production-reachable and carry CVEs
# at older versions. Minimum floors ensure the patched version is always resolved.

# certifi: multiple root-cert CVEs. Patched in 2026.10.23.
certifi>=2026.10.23

# pillow: multiple image-processing CVEs. Patched in 13.0.0 (reportlab>=4.0.0
# has no upper bound on pillow, so the major bump is safe).
pillow>=13.0.0

# urllib3: PYSEC-2026-4175/4176/4177. Patched in 2.8.0. Pulled in via
# requests and sentry-sdk, both direct production deps.
urllib3>=2.8.0

# anyio: pulled in via starlette/httpx. anyio <= 4.14.x's pytest plugin
# imports _pytest.python.CallSpec2, which pytest 9.2 deprecates; with
# filterwarnings = error that aborts collection. 4.15.0 imports CallSpec
# first (agronholm/anyio, pytest-dev/pytest#14742). Floor only, no cap.
anyio>=4.15.0

# Dev-only transitive package floors (pulled in via pre-commit 4.0.1).
# These are NOT in the production image but are audited in the full-venv
# informational pip-audit step.
virtualenv>=21.14.9
filelock>=4.2.1
msgpack>=1.2.3
setuptools>=84.1.0
